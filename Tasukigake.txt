<script>
    let activeId = 'p';
    let currentMode = 'tasuki';
    let selectedLevels = [2, 3];
    let currentPlayLevel = 2;
    let currentGoal = { p:0, q:0, r:0, s:0, a:0, b:0, c:0 };
    let gameState = 'playing'; 
    let startTime, timerInterval;

    let cfgP = 7, cfgQ = 9, cfgSafeOn = true, cfgSafeA = 40, cfgSafeC = 60;

    let isMouseDownOutside = false;
    const modalOverlay = document.getElementById('level-modal');

    modalOverlay.addEventListener('mousedown', (e) => { isMouseDownOutside = (e.target === modalOverlay); });
    modalOverlay.addEventListener('mouseup', (e) => {
        if (isMouseDownOutside && e.target === modalOverlay) toggleModal(false);
        isMouseDownOutside = false;
    });

    function toggleModal(show) {
        modalOverlay.style.display = show ? 'flex' : 'none';
        if(show) {
            document.querySelectorAll('.lvl-cb').forEach(cb => {
                cb.checked = selectedLevels.includes(parseInt(cb.value));
            });
            updateMaxExample();
        }
    }

    function resetAdvancedSettings() {
        document.getElementById('sl-maxp').value = 7;
        document.getElementById('sl-maxq').value = 9;
        document.getElementById('cb-safety').checked = true;
        document.getElementById('sl-safety-a').value = 40;
        document.getElementById('sl-safety-c').value = 60;
        ['maxp', 'maxq', 'safety-a', 'safety-c'].forEach(id => { 
            document.getElementById('sl-'+id).dispatchEvent(new Event('input')); 
        });
        document.getElementById('cb-safety').dispatchEvent(new Event('change'));
    }

    ['maxp', 'maxq', 'safety-a', 'safety-c'].forEach(id => {
        document.getElementById('sl-' + id).addEventListener('input', (e) => {
            document.getElementById('lbl-' + id).innerText = e.target.value;
            updateMaxExample();
        });
    });

    document.getElementById('cb-safety').addEventListener('change', (e) => {
        const isOn = e.target.checked;
        ['a', 'c'].forEach(suffix => {
            const row = document.getElementById('row-safety-' + suffix);
            row.style.opacity = isOn ? "1" : "0.4";
            row.querySelectorAll('input').forEach(i => i.disabled = !isOn);
        });
        updateMaxExample();
    });

    function updateMaxExample() {
        const pM = parseInt(document.getElementById('sl-maxp').value);
        const qM = parseInt(document.getElementById('sl-maxq').value);
        const safeOn = document.getElementById('cb-safety').checked;
        const sAM = parseInt(document.getElementById('sl-safety-a').value);
        const sCM = parseInt(document.getElementById('sl-safety-c').value);

        let bestStr = ""; let maxB = -1;
        for(let p=pM; p>=1; p--) {
            for(let r=pM; r>=1; r--) {
                let a = p*r; if(safeOn && a > sAM) continue;
                for(let q=qM; q>=1; q--) {
                    for(let s=qM; s>=1; s--) {
                        let c = q*s; if(safeOn && Math.abs(c) > sCM) continue;
                        let b = p*s + r*q;
                        if(Math.abs(b) > maxB) { maxB = Math.abs(b); bestStr = `${a}x² ${b>=0?'+':'-'} ${Math.abs(b)}x ${c>=0?'+':'-'} ${Math.abs(c)}`; }
                    }
                }
            }
            if(bestStr) break; 
        }
        document.getElementById('max-example-disp').innerText = bestStr || "Searching...";
    }

    function applyLevels() {
        const checked = Array.from(document.querySelectorAll('.lvl-cb:checked')).map(cb => parseInt(cb.value));
        if (checked.length === 0) return;
        selectedLevels = checked;
        cfgP = parseInt(document.getElementById('sl-maxp').value);
        cfgQ = parseInt(document.getElementById('sl-maxq').value);
        cfgSafeOn = document.getElementById('cb-safety').checked;
        cfgSafeA = parseInt(document.getElementById('sl-safety-a').value);
        cfgSafeC = parseInt(document.getElementById('sl-safety-c').value);
        toggleModal(false);
        newGame();
    }

    function isPrime(num) {
        let n = Math.abs(num); if (n <= 1) return false;
        for (let i = 2; i <= Math.sqrt(n); i++) if (n % i === 0) return false;
        return true;
    }
    const rand = (min, max) => Math.floor(Math.random() * (max - min + 1)) + min;
    const gcd = (x, y) => y === 0 ? x : gcd(y, x % y);

    function newGame() {
        gameState = 'playing';
        document.getElementById('btn-submit').innerText = "Submit";
        document.getElementById('btn-giveup').innerText = "Give Up";
        document.getElementById('msg').innerText = "";
        ['p','q','r','s','p2','q2','r2','s2'].forEach(id => document.getElementById(id).classList.remove('giveup-text'));

        currentPlayLevel = selectedLevels[Math.floor(Math.random() * selectedLevels.length)];
        renderLevelDisplay();

        let p, q, r, s, a, b, c;
        while (true) {
            p = rand(1, cfgP); r = rand(1, cfgP); q = rand(-cfgQ, cfgQ); s = rand(-cfgQ, cfgQ);
            if (q === 0 || s === 0 || p*s + r*q === 0) continue;
            if (gcd(p, Math.abs(q)) !== 1 || gcd(r, Math.abs(s)) !== 1) continue;
            a = p * r; c = q * s; b = p * s + r * q;
            if(cfgSafeOn && (a > cfgSafeA || Math.abs(c) > cfgSafeC)) continue;

            let absC = Math.abs(c);
            let aP = isPrime(a), cP = isPrime(absC);
            let valid = false;
            if (currentPlayLevel === 1) valid = (a === 1 || absC === 1);
            else if (currentPlayLevel === 2) valid = (aP && cP);
            else if (currentPlayLevel === 3) valid = (aP || cP);
            else if (currentPlayLevel === 4) valid = (!aP && a>1 && !cP && absC>1);
            if (valid) break;
        }
        currentGoal = { p, q, r, s, a, b, c };
        
        const format = (n, x, isFirst) => {
            let sgn = n > 0 ? (isFirst ? "" : "+ ") : (isFirst ? "-" : "- ");
            let val = Math.abs(n) === 1 && x !== "" ? "" : Math.abs(n);
            return sgn + val + x + " ";
        };
        document.getElementById('display-eq').innerText = (format(a, "x²", true) + format(b, "x", false) + format(c, "", false)).trim();
        
        clearAll();
        setFocus(currentMode === 'tasuki' ? 'p' : 'p2');
        startTimer();
        setTimeout(drawTasukiLines, 50); 
    }

    function renderLevelDisplay() {
        const container = document.getElementById('level-display');
        container.innerHTML = '';
        selectedLevels.sort().forEach(lv => {
            const span = document.createElement('span');
            span.className = 'lvl-item' + (lv === currentPlayLevel ? ' current' : '');
            span.innerText = `Lv.${lv}`;
            container.appendChild(span);
        });
    }

    const navMap = {
        tasuki: { p:{right:'q',down:'r'}, q:{left:'p',down:'s'}, r:{up:'p',right:'s'}, s:{up:'q',left:'r'} },
        formula: { p2:{right:'q2',down:'r2'}, q2:{left:'p2',down:'s2'}, r2:{up:'p2',right:'s2'}, s2:{up:'q2',left:'r2'} }
    };

    document.addEventListener('keydown', (e) => {
        if (["ArrowUp","ArrowDown","ArrowLeft","ArrowRight","Enter","Backspace"].includes(e.key)) e.preventDefault();
        if (e.key >= '0' && e.key <= '9') typeIn(e.key);
        else if (e.key === '-') typeIn('-');
        else if (e.key === 'Backspace') backspace();
        else if (e.key === 'Enter') (gameState === 'playing' ? handleSubmit() : newGame());
        else if (e.key.startsWith('Arrow')) {
            const dir = e.key.replace('Arrow', '').toLowerCase();
            const nextId = navMap[currentMode][activeId][dir];
            if (nextId) setFocus(nextId);
        }
    });

    function setFocus(id) {
        document.querySelectorAll('.input-box').forEach(b => b.classList.remove('active'));
        activeId = id;
        document.getElementById(id).classList.add('active');
    }

    function typeIn(v) {
        if(gameState !== 'playing') return;
        const el = document.getElementById(activeId);
        if (v === '-') {
            if(['p','r','p2','r2'].includes(activeId)) return;
            el.innerText = el.innerText.startsWith('-') ? el.innerText.slice(1) : '-' + el.innerText;
        } else {
            if (el.innerText.replace('-','').length < 3) el.innerText += v;
        }
        syncInputs(); refreshCalc();
    }

    function syncInputs() {
        const base = activeId.replace('2','');
        const target = activeId.includes('2') ? base : base + '2';
        document.getElementById(target).innerText = document.getElementById(activeId).innerText;
    }

    function backspace() {
        if(gameState !== 'playing') return;
        const el = document.getElementById(activeId);
        el.innerText = el.innerText.slice(0, -1);
        syncInputs(); refreshCalc();
    }

    function clearAll() {
        ['p','q','r','s','p2','q2','r2','s2'].forEach(id => document.getElementById(id).innerText = "");
        refreshCalc();
    }

    function refreshCalc() {
        const show = document.getElementById('show-hint').checked;
        const p = parseInt(document.getElementById('p').innerText) || 0;
        const q = parseInt(document.getElementById('q').innerText) || 0;
        const r = parseInt(document.getElementById('r').innerText) || 0;
        const s = parseInt(document.getElementById('s').innerText) || 0;
        const set = (id, val, suf="") => document.getElementById(id).innerText = show ? (val+suf) : "?"+suf;
        set('hint-rq', r*q, "x"); set('hint-ps', p*s, "x");
        set('hint-b', (r*q)+(p*s), "x"); set('hint-a', p*r, "x²"); set('hint-c', q*s);
    }

    function handleGiveUp() {
        if (gameState !== 'playing') return newGame();
        stopTimer(); gameState = 'giveup';
        const setAns = (id, val) => { const el=document.getElementById(id); el.innerText=val; el.classList.add('giveup-text'); };
        setAns('p',currentGoal.p); setAns('p2',currentGoal.p); setAns('q',currentGoal.q); setAns('q2',currentGoal.q);
        setAns('r',currentGoal.r); setAns('r2',currentGoal.r); setAns('s',currentGoal.s); setAns('s2',currentGoal.s);
        refreshCalc();
        document.getElementById('msg').innerText = "Answer"; document.getElementById('msg').style.color = "#EF4444";
        document.getElementById('btn-giveup').innerText = "Next";
    }

    function handleSubmit() {
        if (gameState !== 'playing') return newGame();
        const p = parseInt(document.getElementById('p').innerText), q = parseInt(document.getElementById('q').innerText);
        const r = parseInt(document.getElementById('r').innerText), s = parseInt(document.getElementById('s').innerText);
        
        // --- FIXED: Explicitly set red color for "Fill all boxes" ---
        if ([p,q,r,s].some(isNaN)) { 
            const msgEl = document.getElementById('msg');
            msgEl.innerText = "Fill all boxes"; 
            msgEl.style.color = "#EF4444"; // Always Red
            return; 
        }

        const ok = (p*r===currentGoal.a && q*s===currentGoal.c && p*s+r*q===currentGoal.b) || (r*p===currentGoal.a && s*q===currentGoal.c && r*q+p*s===currentGoal.b);
        if (ok) {
            stopTimer(); gameState = 'correct';
            document.getElementById('msg').innerText = "Correct!"; document.getElementById('msg').style.color = "#10B981";
            document.getElementById('btn-submit').innerText = "Next";
        } else {
            document.getElementById('msg').innerText = "Incorrect!"; document.getElementById('msg').style.color = "#EF4444";
            document.getElementById('game-card').style.animation = 'shake 0.4s';
            setTimeout(() => document.getElementById('game-card').style.animation = '', 400);
        }
    }

    function setMode(mode) {
        currentMode = mode;
        document.getElementById('layout-tasuki').style.display = mode === 'tasuki' ? 'grid' : 'none';
        document.getElementById('layout-formula').style.display = mode === 'formula' ? 'flex' : 'none';
        document.getElementById('mode-tasuki').classList.toggle('active', mode === 'tasuki');
        document.getElementById('mode-formula').classList.toggle('active', mode === 'formula');
        setFocus(mode === 'tasuki' ? activeId.replace('2','') : activeId.replace('2','')+'2');
        if(mode === 'tasuki') setTimeout(drawTasukiLines, 10);
    }

    function drawTasukiLines() {
        const svg = document.getElementById('tasuki-svg');
        const getC = (id) => { const el = document.getElementById(id); return { x: el.offsetLeft + el.offsetWidth/2, y: el.offsetTop + el.offsetHeight/2 }; };
        const pC = getC('p'), qC = getC('q'), rC = getC('r'), sC = getC('s');
        svg.innerHTML = `<line x1="${pC.x}" y1="${pC.y}" x2="${sC.x}" y2="${sC.y}" stroke="#94A3B8" stroke-width="1.5" /><line x1="${rC.x}" y1="${rC.y}" x2="${qC.x}" y2="${qC.y}" stroke="#94A3B8" stroke-width="1.5" />`;
    }

    function toggleFullScreen() {
        if (!document.fullscreenElement) document.documentElement.requestFullscreen();
        else if (document.exitFullscreen) document.exitFullscreen();
        setTimeout(drawTasukiLines, 100);
    }

    function startTimer() { startTime = Date.now(); clearInterval(timerInterval); timerInterval = setInterval(() => { document.getElementById('timer-display').innerText = ((Date.now() - startTime) / 1000).toFixed(2) + 's'; }, 10); }
    function stopTimer() { clearInterval(timerInterval); }

    window.onload = newGame;
    window.addEventListener('resize', drawTasukiLines);
